#!/bin/bash -l
#SBATCH --job-name=ICON.run
#SBATCH --partition=hpc-mpi
#SBATCH --output=output.txt
#SBATCH --error=error.txt
#SBATCH --nodes=1
#SBATCH --ntasks-per-node=32
#SBATCH --ntasks-per-core=1
#SBATCH --time=30:00
#SBATCH --hint=nomultithread

# Load required modules
source /home/$USER/.commonrc
gogoicon

# Prevent memory crashes
ulimit -s unlimited
export OMP_STACKSIZE=512M
export KMP_STACKSIZE=512M
export I_MPI_THREAD_STACK_SIZE=512M

# Directory setup
export OUTDIR="/shared/scratch/$USER/simulations/output"
mkdir -p $OUTDIR
cd $OUTDIR

# Execute the binary
ICON_BIN="/home/$USER/Models/icon-model/build_intel/bin/icon"
mpirun -np $SLURM_NTASKS $ICON_BIN
